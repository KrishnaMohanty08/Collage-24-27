# Write a Python program to demonstrate data preprocessing steps: handling missing values, encoding categorical data, and feature scaling.
import numpy as np 
import pandas as pd

df=pd.read_csv("./Datasets/IRIS-DATA.csv")
print("sample 5 data",df.head())


#Check this column for NaN → None exist → nothing changes
cols=['sepal_length','sepal_width','petal_length','petal_width']
#If NaN existed, they would be replaced with mean(sepal_length).
for col in cols:
    df[col]=df[col].fillna(df[col].mean()) 

# Fill categorical column with mode
df['species']=df ['species'].fillna(df['species'].mode()[0])


# A. Label Encoding Manually
species_unique = df['species'].unique()
#df['species'].unique() -This returns all unique values from the species column.
label_mapping = {name: idx for idx, name in enumerate(species_unique)}
#enumerate(species_unique) -assigns :{key = species name,value = number label}


df['species_label'] = df['species'].map(label_mapping)
print("\nLabel Encoding Mapping:", label_mapping)
print("\nAfter Label Encoding:\n", df.head())


# A. Standardization (Z-score)
'''
Subtract the column mean
Divide by the column standard deviation

Formula:
z =(x-mean)/std
'''
for col in cols:
    df[col + "_std"] = (df[col] - df[col].mean()) / df[col].std()

# B. Min-Max Normalization
for col in cols:
    df[col + "_norm"] = (df[col] - df[col].min()) / (df[col].max() - df[col].min())

print("\nAfter Feature Scaling:\n", df.head())

print("\nPreprocessing Completed Successfully!")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df=pd.read_csv("./Datasets/height-weight.csv")
print("sample 5 data",df.head())

height=df['Height']
weight=df['Weight']

def Skewness(n, data):
    mean = np.mean(data)
    std = np.std(data, ddof=1)
    skewness = np.sum((data - mean) ** 3) / (n * (std ** 3))
    return skewness

# Call the function with parameters
print("skewness of height:", Skewness(len(height), height))
print("skewness of weight:", Skewness(len(weight), weight))


plt.figure(figsize=(12, 5))

plt.subplot(1, 2, 1)
plt.hist(height, bins=30, color='blue', alpha=0.7, edgecolor='black')
plt.xlabel('Height')
plt.ylabel('Frequency')
plt.title('Height Distribution')

plt.subplot(1, 2, 2)
plt.hist(weight, bins=30, color='green', alpha=0.7, edgecolor='black')
plt.xlabel('Weight')
plt.ylabel('Frequency')
plt.title('Weight Distribution')

plt.tight_layout()
plt.show()
